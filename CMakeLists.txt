cmake_minimum_required(VERSION 3.23)
project(OpenBrailleEmbosser VERSION 0.1.0 LANGUAGES CXX)

include(GNUInstallDirs)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(BRAILLE_CONVERTER "Build document converter" ON)
option(BRAILLE_MUSIC_SCAN "Build native optical music recognition" OFF)
option(BRAILLE_SERVICE "Build Linux upload and IPP service" OFF)
option(BRAILLE_DESKTOP "Build the Qt desktop sender" OFF)
option(BRAILLE_BENCH "Build the Qt UNO Q signal bench" OFF)
set(BRAILLE_PAPPL_ROOT "" CACHE PATH "Prefix of the patched PAPPL build")
set(BRAILLE_ONNXRUNTIME_ROOT "" CACHE PATH "Prefix containing the ONNX Runtime headers and library")

if(BRAILLE_CONVERTER)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(CONVERTER REQUIRED IMPORTED_TARGET liblouis icu-uc libxml-2.0 libzip lept tesseract)
    pkg_check_modules(MUPDF REQUIRED mupdf)
    add_library(braille_core STATIC src/braille.cpp src/docx.cpp src/imaging.cpp src/music.cpp src/pdf.cpp src/picture.cpp)
    target_include_directories(braille_core PUBLIC src PRIVATE ${MUPDF_INCLUDE_DIRS})
    target_link_libraries(braille_core PRIVATE PkgConfig::CONVERTER ${MUPDF_STATIC_LDFLAGS} harfbuzz)
    target_compile_definitions(braille_core PRIVATE
        BRAILLE_SOURCE_TABLE="${CMAKE_CURRENT_SOURCE_DIR}/spanish-numbers.cti"
        BRAILLE_RELATIVE_TABLE="${CMAKE_INSTALL_DATADIR}/open-braille-embosser/spanish-numbers.cti"
        BRAILLE_INSTALLED_TABLE="${CMAKE_INSTALL_FULL_DATADIR}/open-braille-embosser/spanish-numbers.cti")
    add_executable(braille src/cli.cpp)
    target_link_libraries(braille PRIVATE braille_core)
    add_executable(pdf-render src/pdf_render_cli.cpp)
    target_link_libraries(pdf-render PRIVATE braille_core)
    install(TARGETS braille pdf-render)
    install(FILES spanish-numbers.cti DESTINATION "${CMAKE_INSTALL_DATADIR}/open-braille-embosser")
endif()

if(BRAILLE_MUSIC_SCAN)
    if(NOT BRAILLE_CONVERTER)
        message(FATAL_ERROR "BRAILLE_MUSIC_SCAN requires the converter")
    endif()
    find_path(ONNXRUNTIME_INCLUDE onnxruntime_cxx_api.h HINTS "${BRAILLE_ONNXRUNTIME_ROOT}/include" REQUIRED)
    find_library(ONNXRUNTIME_LIBRARY onnxruntime HINTS "${BRAILLE_ONNXRUNTIME_ROOT}/lib" REQUIRED)
    target_sources(braille PRIVATE src/omr.cpp src/omr_model.cpp)
    target_include_directories(braille PRIVATE "${ONNXRUNTIME_INCLUDE}")
    target_link_libraries(braille PRIVATE "${ONNXRUNTIME_LIBRARY}" PkgConfig::CONVERTER)
    target_compile_definitions(braille PRIVATE BRAILLE_MUSIC_SCAN)
endif()

if(BUILD_TESTING)
    add_executable(test_firmware tests/native/firmware.cpp)
    target_include_directories(test_firmware PRIVATE firmware/uno_q)

    add_executable(test_actuator tests/native/actuator.cpp)

        "${CMAKE_CURRENT_BINARY_DIR}/test-products/actuators")

if(BRAILLE_BENCH)
    add_subdirectory(bench)
endif()

if(BRAILLE_SERVICE)
    if(NOT BRAILLE_CONVERTER OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "BRAILLE_SERVICE requires the converter on Linux")
    endif()
    pkg_check_modules(CUPS REQUIRED IMPORTED_TARGET cups)
    pkg_check_modules(HTTP REQUIRED IMPORTED_TARGET cpp-httplib)
    pkg_check_modules(AVAHI REQUIRED IMPORTED_TARGET avahi-client)
    find_package(OpenSSL REQUIRED)
    find_package(Threads REQUIRED)
    find_package(nlohmann_json 3.10 REQUIRED)
    if(NOT BRAILLE_PAPPL_ROOT)
        message(FATAL_ERROR "Set BRAILLE_PAPPL_ROOT to the patched PAPPL 1.3.1 prefix. See README.md.")
    endif()
    find_library(PAPPL_LIBRARY pappl PATHS "${BRAILLE_PAPPL_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    add_library(braille_raster STATIC src/raster.cpp)
    target_link_libraries(braille_raster PUBLIC braille_core PkgConfig::CUPS)
    add_executable(braille-printer src/printer.cpp)
    target_include_directories(braille-printer PRIVATE "${BRAILLE_PAPPL_ROOT}/include")
    target_link_libraries(braille-printer PRIVATE braille_raster "${PAPPL_LIBRARY}")
    set_property(TARGET braille-printer PROPERTY INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
    add_library(braille_jobs STATIC src/job_settings.cpp)
    target_link_libraries(braille_jobs PUBLIC braille_core nlohmann_json::nlohmann_json)
    add_executable(braille-service src/service.cpp)
    target_include_directories(braille-service PRIVATE firmware/uno_q)
    target_link_libraries(braille-service PRIVATE braille_raster braille_jobs PkgConfig::HTTP PkgConfig::AVAHI
        nlohmann_json::nlohmann_json OpenSSL::Crypto Threads::Threads)
    install(TARGETS braille-service braille-printer)
    install(FILES "${BRAILLE_PAPPL_ROOT}/lib/libpappl.so.1" DESTINATION "${CMAKE_INSTALL_LIBDIR}")
    install(DIRECTORY "${BRAILLE_PAPPL_ROOT}/share/pappl" DESTINATION "${CMAKE_INSTALL_DATADIR}" OPTIONAL)
    if(BUILD_TESTING)
        # service: job settings and the conversion worker. printer: the browser
        # forms, IPP queue and raster decoding against a running braille-service.
        foreach(suite IN ITEMS service printer)
            add_executable(test_${suite} tests/native/${suite}.cpp)
            target_include_directories(test_${suite} PRIVATE tests/native ${MUPDF_INCLUDE_DIRS})
            target_link_libraries(test_${suite} PRIVATE braille_jobs braille_raster PkgConfig::CONVERTER ${MUPDF_STATIC_LDFLAGS})
            add_test(NAME ${suite} COMMAND test_${suite} "${CMAKE_CURRENT_SOURCE_DIR}"
                "${CMAKE_CURRENT_BINARY_DIR}/test-products/${suite}" "$<TARGET_FILE:braille-service>")
            set_tests_properties(${suite} PROPERTIES ENVIRONMENT "OMP_THREAD_LIMIT=1" TIMEOUT 300)
        endforeach()
        target_link_libraries(test_printer PRIVATE PkgConfig::HTTP Threads::Threads)
    endif()
endif()

if(BRAILLE_CONVERTER AND UNIX)
    add_executable(braille-demo demo/build_demo.cpp)
    target_include_directories(braille-demo PRIVATE ${MUPDF_INCLUDE_DIRS})
    target_link_libraries(braille-demo PRIVATE braille_core)
    # `cmake --build build --target demo` runs every demo conversion and writes
    # the presentation. The builder refuses destinations inside the source tree
    # other than output/, so the default is in the temporary directory.
    set(demo_temporary "$ENV{TMPDIR}")
    if(NOT demo_temporary)
        set(demo_temporary /tmp)
    endif()
    set(BRAILLE_DEMO_OUTPUT "${demo_temporary}/open-braille-demo/Software Demo.pdf" CACHE FILEPATH
        "Presentation PDF written by the demo target")
    set(demo_work "${CMAKE_CURRENT_BINARY_DIR}/demo-work")
    add_custom_target(demo
        COMMAND "${CMAKE_COMMAND}" -E rm -rf "${demo_work}"
        COMMAND "${CMAKE_COMMAND}" -E env OMP_THREAD_LIMIT=1 "$<TARGET_FILE:braille-demo>"
            "${CMAKE_CURRENT_SOURCE_DIR}" "$<TARGET_FILE:braille>" "${demo_work}" "${BRAILLE_DEMO_OUTPUT}"
        USES_TERMINAL VERBATIM)
    add_dependencies(demo braille braille-demo)
endif()

option(BRAILLE_PACKAGE_SENDER "Stage one distributable desktop sender archive" OFF)
set(BRAILLE_QT_PLUGINS "" CACHE PATH "Qt plugin directory used by Linux desktop packaging")
set(BRAILLE_SENDER_NOTICES "" CACHE PATH "Notices and corresponding-source links for the Qt runtime being packaged")

if(BRAILLE_PACKAGE_SENDER)
    if(NOT BRAILLE_DESKTOP OR NOT IS_DIRECTORY "${BRAILLE_SENDER_NOTICES}")
        message(FATAL_ERROR "Enable BRAILLE_DESKTOP and provide the bundled runtime notices in BRAILLE_SENDER_NOTICES")
    endif()
    set(sender_bin .)
    if(WIN32)
        set(sender_bin "${CMAKE_INSTALL_BINDIR}")
    endif()
    install(TARGETS braille-sender RUNTIME DESTINATION "${sender_bin}" COMPONENT Sender BUNDLE DESTINATION . COMPONENT Sender)
    install(FILES LICENSE DESTINATION . COMPONENT Sender)
    install(DIRECTORY "${BRAILLE_SENDER_NOTICES}/" DESTINATION licenses COMPONENT Sender)
    if(APPLE OR WIN32)
        if(Qt6_VERSION VERSION_LESS 6.5)
            qt_generate_deploy_app_script(TARGET braille-sender FILENAME_VARIABLE deploy_sender)
        else()
            qt_generate_deploy_app_script(TARGET braille-sender OUTPUT_SCRIPT deploy_sender)
        endif()
        install(SCRIPT "${deploy_sender}" COMPONENT Sender)
        set(CPACK_GENERATOR ZIP)
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        get_target_property(qt_core Qt6::Core IMPORTED_LOCATION_RELEASE)
        get_filename_component(sender_qt_libdir "${qt_core}" DIRECTORY)
        if(NOT BRAILLE_QT_PLUGINS)
            find_path(BRAILLE_QT_PLUGINS_DEFAULT NAMES platforms/libqxcb.so PATHS
                "${sender_qt_libdir}/qt6/plugins" "${sender_qt_libdir}/../plugins" NO_DEFAULT_PATH REQUIRED)
            set(BRAILLE_QT_PLUGINS "${BRAILLE_QT_PLUGINS_DEFAULT}")
        endif()
        # The executable finds the bundled Qt in lib/ through its RPATH, and
        # qt.conf points Qt at the bundled plugins/. An old-style DT_RPATH, unlike
        # DT_RUNPATH, also resolves the dependencies of those libraries and plugins.
        set_target_properties(braille-sender PROPERTIES INSTALL_RPATH "$ORIGIN/lib" BUILD_WITH_INSTALL_RPATH ON)
        target_link_options(braille-sender PRIVATE "LINKER:--disable-new-dtags")
        configure_file(cmake/StageSenderLinux.cmake.in stage-sender-linux.cmake @ONLY)
        install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/stage-sender-linux.cmake" COMPONENT Sender)
        install(FILES desktop/qt.conf DESTINATION . COMPONENT Sender)
        set(CPACK_GENERATOR TGZ)
    else()
        message(FATAL_ERROR "Sender packaging targets Linux, macOS, and Windows")
    endif()
    set(CPACK_PACKAGE_NAME Open-Braille-Sender)
    set(sender_arch "${CMAKE_SYSTEM_PROCESSOR}")
    if(APPLE AND CMAKE_OSX_ARCHITECTURES)
        string(REPLACE ";" "-" sender_arch "${CMAKE_OSX_ARCHITECTURES}")
    elseif(WIN32 AND CMAKE_GENERATOR_PLATFORM)
        set(sender_arch "${CMAKE_GENERATOR_PLATFORM}")
    endif()
    set(CPACK_PACKAGE_FILE_NAME "Open-Braille-Sender-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${sender_arch}")
    set(CPACK_COMPONENTS_ALL Sender)
    set(CPACK_ARCHIVE_COMPONENT_INSTALL ON)
    set(CPACK_COMPONENT_INCLUDE_TOPLEVEL_DIRECTORY ON)
    if(CPACK_GENERATOR STREQUAL "TGZ")
        set(CPACK_ARCHIVE_SENDER_FILE_NAME "${CPACK_PACKAGE_FILE_NAME}")
    endif()
    include(CPack)
endif()
